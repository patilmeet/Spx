create table if not exists ghost_devices (
  device_id   text primary key,
  info        text,
  battery     text,
  last_seen   timestamptz default now(),
  online      boolean default false,
  live_active boolean default false
);

create table if not exists ghost_commands (
  id         bigserial primary key,
  device_id  text not null,
  command    text not null,
  param      text default '',
  executed   boolean default false,
  created_at timestamptz default now()
);

create table if not exists ghost_results (
  id         bigserial primary key,
  device_id  text not null,
  type       text not null,
  payload    text,
  created_at timestamptz default now()
);

create table if not exists ghost_live_frames (
  device_id  text primary key,
  facing     text default 'environment',
  frame      text,
  seq        bigint default 0,
  w          integer,
  h          integer,
  ts         timestamptz default now()
);

alter table ghost_devices     enable row level security;
alter table ghost_commands    enable row level security;
alter table ghost_results     enable row level security;
alter table ghost_live_frames enable row level security;

create policy "anon all" on ghost_devices     for all using (true) with check (true);
create policy "anon all" on ghost_commands    for all using (true) with check (true);
create policy "anon all" on ghost_results     for all using (true) with check (true);
create policy "anon all" on ghost_live_frames for all using (true) with check (true);
